/* ===== GIỌNG ĐỌC MẪU (TTS) ===== */
let femaleVoice = null;
function loadVoices(){
  const voices = speechSynthesis.getVoices();
  femaleVoice = voices.find(v=>v.name.includes("Google US English")) ||
                voices.find(v=>v.name.includes("Microsoft Aria")) ||
                voices.find(v=>v.name.includes("Samantha")) ||
                voices.find(v=>v.name.includes("Female")) ||
                voices.find(v=>v.lang==="en-US") ||
                voices.find(v=>v.lang.startsWith("en"));
}
loadVoices();
if(speechSynthesis.onvoiceschanged !== undefined){ speechSynthesis.onvoiceschanged = loadVoices; }

function speak(text){
  speechSynthesis.cancel();
  const utter = new SpeechSynthesisUtterance(text);
  utter.lang = "en-US";
  utter.rate = 0.85;
  utter.pitch = 1.15;
  if(femaleVoice){ utter.voice = femaleVoice; }
  speechSynthesis.speak(utter);
}

function speakSentence(text){
  speechSynthesis.cancel();
  const msg = new SpeechSynthesisUtterance(text);
  msg.lang = "en-US";
  msg.rate = 0.85;
  msg.pitch = 1.2;
  if(femaleVoice){ msg.voice = femaleVoice; }
  speechSynthesis.speak(msg);
}

/* ===== HỆ THỐNG GHI ÂM VÀ BỘ ĐẾM 60S ===== */
let mediaRecorder = null;
let audioChunks = [];
let activeTimer = null;
let recordedStatus = {}; // Lưu trạng thái hoàn thành ghi âm từng nút

document.addEventListener("DOMContentLoaded", function() {
  initRecorderUI();
});

function initRecorderUI() {
  // Tìm tất cả các nút nghe có sẵn
  const audioElements = document.querySelectorAll(".audio, .audio-word, .speak-btn");
  
  audioElements.forEach((el, index) => {
    // Bọc trong container chung
    const wrapper = document.createElement("div");
    wrapper.className = "audio-group-wrapper";
    
    el.parentNode.insertBefore(wrapper, el);
    wrapper.appendChild(el);

    // Tạo nút Record & Timer 60s
    const recBtn = document.createElement("button");
    recBtn.className = "rec-btn";
    recBtn.id = `rec-btn-${index}`;
    recBtn.innerHTML = `🎙️ <span class="timer-circle" id="timer-${index}">60</span>`;
    recBtn.onclick = () => toggleRecording(index);
    
    // Nút Nghe lại (Mặc định ẩn)
    const playBtn = document.createElement("button");
    playBtn.className = "play-rec-btn hidden";
    playBtn.id = `play-btn-${index}`;
    playBtn.innerHTML = `▶ Nghe lại`;

    wrapper.appendChild(recBtn);
    wrapper.appendChild(playBtn);
  });

  // Cập nhật tổng số nút ghi âm lên thanh tiến độ
  const total = document.querySelectorAll(".audio-group-wrapper").length;
  document.getElementById("totalCount").innerText = total;
  updateProgressBar();
}

async function toggleRecording(index) {
  const recBtn = document.getElementById(`rec-btn-${index}`);
  const timerEl = document.getElementById(`timer-${index}`);
  const playBtn = document.getElementById(`play-btn-${index}`);

  // Nếu đang ghi âm ở nút này -> Bấm dừng
  if (recBtn.classList.contains("recording")) {
    stopRecording(index);
    return;
  }

  // Xin quyền Microphone
  try {
    const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
    
    // Dừng bất kỳ hoạt động ghi âm trước đó nếu có
    if(mediaRecorder && mediaRecorder.state === "recording") {
      mediaRecorder.stop();
    }
    clearInterval(activeTimer);

    audioChunks = [];
    mediaRecorder = new MediaRecorder(stream);
    
    mediaRecorder.ondataavailable = e => {
      if (e.data.size > 0) audioChunks.push(e.data);
    };

    mediaRecorder.onstop = () => {
      const audioBlob = new Blob(audioChunks, { type: 'audio/mp3' });
      const audioUrl = URL.createObjectURL(audioBlob);
      
      playBtn.onclick = () => {
        const audio = new Audio(audioUrl);
        audio.play();
      };
      playBtn.classList.remove("hidden");

      // Đánh dấu đã ghi âm xong nút này
      recordedStatus[index] = true;
      updateProgressBar();
      checkAllCompleted();
    };

    mediaRecorder.start();
    recBtn.classList.add("recording");

    // Chạy đếm ngược 60 giây
    let timeLeft = 60;
    timerEl.innerText = timeLeft;

    activeTimer = setInterval(() => {
      timeLeft--;
      timerEl.innerText = timeLeft;
      if (timeLeft <= 0) {
        stopRecording(index);
      }
    }, 1000);

  } catch (err) {
    alert("Không thể truy cập Microphone! Vui lòng cấp quyền micro cho trình duyệt.");
  }
}

function stopRecording(index) {
  const recBtn = document.getElementById(`rec-btn-${index}`);
  const timerEl = document.getElementById(`timer-${index}`);
  
  if (mediaRecorder && mediaRecorder.state === "recording") {
    mediaRecorder.stop();
    mediaRecorder.stream.getTracks().forEach(track => track.stop());
  }
  clearInterval(activeTimer);
  
  recBtn.classList.remove("recording");
  timerEl.innerText = "60";
}

/* ===== CẬP NHẬT THANH TIẾN ĐỘ ===== */
function updateProgressBar() {
  const totalBtns = document.querySelectorAll(".audio-group-wrapper").length;
  const completedCount = Object.keys(recordedStatus).length;
  const percentage = totalBtns > 0 ? Math.round((completedCount / totalBtns) * 100) : 0;

  document.getElementById("completedCount").innerText = completedCount;
  document.getElementById("progressText").innerText = `${percentage}%`;
  document.getElementById("progressBar").style.width = `${percentage}%`;
}

    }, 500);
  }
}
